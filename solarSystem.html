Description

This Solar System project introduces the planets, moons, and basic space science concepts. It explains how the system is organized, how the planets move, and what makes Earth unique among the planets.

Method

Step 1: Download Textures

Go to solarsystemscope.com/textures and download all required texture files. Save them in one folder (e.g. C:/Users/Joseph/Downloads/textures/).

Step 2: Set Up Blender

1. Open Blender
2. Go to the Scripting tab
3. Click New to create a script
4. Paste the Python script into the editor
5. Update texture path:
TEX = "C:/Users/Joseph/Downloads/textures/"

Step 3: Run the Script

1. Click Run Script or press Alt + P
2. Wait for “Done!” message
3. The scene will generate planets, Sun, Moon, rings, and stars

Step 4: Check the Scene

1. Press Numpad 0 for camera view
2. Press Space to preview animation
3. Verify all planets are visible and moving

Step 5: Render the Animation

1. Go to Render menu
2. Click Render Animation or Ctrl + F12
3. Blender renders 250 PNG frames
4. Wait for rendering to finish

Step 6: Find Rendered Frames

Default location: C:\tmp\
Files: 0001.png to 0250.png

If missing:
1. Go to Properties → Output tab
2. Check Output Path

Step 7: Convert to Video (FFmpeg)

Run this command in Command Prompt: 

ffmpeg -framerate 24 -i "C:/tmp/%04d.png" -c:v libx264 -pix_fmt yuv420p "C:/Users/Joseph/OneDrive/Desktop/solar_system.mp4"

def make_material(obj, tex_key, emission=False, alpha=1.0):
    path = TEX.get(tex_key, "")
    mat = bpy.data.materials.new(obj.name + "_Mat")
    mat.use_nodes = True
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out = nt.nodes.new("ShaderNodeOutputMaterial")
    tex = nt.nodes.new("ShaderNodeTexImage")
    if path:
        try:
            img = bpy.data.images.load(path, check_existing=True)
            img.colorspace_settings.name = "sRGB"
            tex.image = img
        except Exception as e:
            print(f"  [WARN] {path}: {e}")
    emit = nt.nodes.new("ShaderNodeEmission")
    strength = 4.0 if emission else 1.0
    emit.inputs[1].default_value = strength
    nt.links.new(tex.outputs[0],  emit.inputs[0])
    nt.links.new(emit.outputs[0], out.inputs[0])
    obj.data.materials.clear()
    obj.data.materials.append(mat)
    obj.visible_shadow = False

# ================= ORBIT RING =================
def make_orbit_ring(name, radius, color=(0.2, 0.4, 0.6), parent=None):
    verts  = 128
    coords = [(math.cos(2*math.pi*i/verts)*radius,
               math.sin(2*math.pi*i/verts)*radius, 0.0)
              for i in range(verts)]
    coords.append(coords[0])
    cd = bpy.data.curves.new(name + "_Ring", type='CURVE')
    cd.dimensions   = '3D'
    cd.bevel_depth  = 0.008
    cd.resolution_u = 4
    sp = cd.splines.new('POLY')
    sp.points.add(len(coords) - 1)
    for i, co in enumerate(coords):
        sp.points[i].co = (*co, 1.0)
    obj = bpy.data.objects.new(name + "_Ring", cd)
    bpy.context.collection.objects.link(obj)
    if parent:
        obj.parent = parent
    mat = bpy.data.materials.new(name + "_RingMat")
    mat.use_nodes = True
    mn = mat.node_tree
    for n in list(mn.nodes): mn.nodes.remove(n)
    em = mn.nodes.new("ShaderNodeEmission")
    mo = mn.nodes.new("ShaderNodeOutputMaterial")
    em.inputs[0].default_value = (*color, 1.0)
    em.inputs[1].default_value = 1.2
    mn.links.new(em.outputs[0], mo.inputs[0])
    obj.data.materials.append(mat)
    return obj

# ================= SATURN RINGS =================
# Returns the mesh so the caller can add a counter-spin driver
def add_saturn_rings(planet_obj):
    bpy.ops.mesh.primitive_cylinder_add(
        radius=3.2, depth=0.02, location=(0, 0, 0))
    r = bpy.context.object
    r.name   = "Saturn_Rings_Mesh"
    r.parent = planet_obj
    r.scale.x = 1.3
    r.visible_shadow = False
    mat = bpy.data.materials.new("Saturn_Ring_Mat")
    mat.use_nodes    = True
    mat.blend_method = 'BLEND'
    mat.use_backface_culling = False
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out   = nt.nodes.new("ShaderNodeOutputMaterial")
    coord = nt.nodes.new("ShaderNodeTexCoord")
    mapping = nt.nodes.new("ShaderNodeMapping")
    mapping.inputs["Scale"].default_value[0] = 2.0
    tex   = nt.nodes.new("ShaderNodeTexImage")
    try:
        img = bpy.data.images.load(TEX["rings"], check_existing=True)
        img.colorspace_settings.name = "sRGB"
        tex.image = img
    except Exception:
        pass
    transp = nt.nodes.new("ShaderNodeBsdfTransparent")
    bsdf   = nt.nodes.new("ShaderNodeBsdfPrincipled")
    mix    = nt.nodes.new("ShaderNodeMixShader")
    mix.inputs["Fac"].default_value = 0.35
    nt.links.new(coord.outputs["UV"],       mapping.inputs["Vector"])
    nt.links.new(mapping.outputs["Vector"], tex.inputs["Vector"])
    nt.links.new(tex.outputs["Color"],      bsdf.inputs["Base Color"])
    nt.links.new(transp.outputs["BSDF"],    mix.inputs[1])
    nt.links.new(bsdf.outputs["BSDF"],      mix.inputs[2])
    nt.links.new(mix.outputs["Shader"],     out.inputs["Surface"])
    r.data.materials.append(mat)
    return r

# Returns the mesh so the caller can add a counter-spin driver
def add_uranus_rings(planet_obj):
    bpy.ops.mesh.primitive_torus_add(
        major_radius=2.2,
        minor_radius=0.04,
        major_segments=128,
        minor_segments=6,
        location=(0, 0, 0))
    r = bpy.context.object
    r.name   = "Uranus_Rings_Mesh"
    r.parent = planet_obj
    r.visible_shadow = False
    mat = bpy.data.materials.new("Uranus_Ring_Mat")
    mat.use_nodes    = True
    mat.blend_method = 'BLEND'
    mat.use_backface_culling = False
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out  = nt.nodes.new("ShaderNodeOutputMaterial")
    emit = nt.nodes.new("ShaderNodeEmission")
    emit.inputs[0].default_value = (0.5, 0.55, 0.6, 1.0)
    emit.inputs[1].default_value = 0.5
    nt.links.new(emit.outputs[0], out.inputs[0])
    r.data.materials.append(mat)
    return r

# ================= DRIVER HELPERS =================
def add_location_drivers(obj, dist, orbit_speed, offset_angle=0.0):
    for axis, trig in ((0, "cos"), (1, "sin")):
        obj.driver_remove("location", axis)
        fc  = obj.driver_add("location", axis)
        drv = fc.driver
        drv.type = 'SCRIPTED'
        var = drv.variables.new()
        var.name = "frame"
        var.type = 'SINGLE_PROP'
        t = var.targets[0]
        t.id_type   = 'SCENE'
        t.id        = scene
        t.data_path = "frame_current"
        drv.expression = f"{trig}(frame * {orbit_speed} + {offset_angle}) * {dist}"

def add_rotation_driver(obj, spin_speed, axis=2):
    obj.driver_remove("rotation_euler", axis)
    fc  = obj.driver_add("rotation_euler", axis)
    drv = fc.driver
    drv.type = 'SCRIPTED'
    var = drv.variables.new()
    var.name = "frame"
    var.type = 'SINGLE_PROP'
    t = var.targets[0]
    t.id_type   = 'SCENE'
    t.id        = scene
    t.data_path = "frame_current"
    drv.expression = f"frame * {spin_speed}"

planet_objs = {}

for (pname, pradius, pdist, tilt_deg, orb_N, spin_N, tex_key, offset) in PLANETS:

    make_orbit_ring(pname, pdist)

    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=pradius, segments=64, ring_count=32,
        location=(pdist, 0, 0))
    planet = bpy.context.object
    planet.name = pname
    bpy.ops.object.shade_smooth()

    planet.rotation_mode  = 'XYZ'
    planet.rotation_euler = (0.0, math.radians(tilt_deg), 0.0)

    make_material(planet, tex_key)

    if pname == "Saturn":
        # Counter-spin the rings opposite to the planet's Z spin.
        # This keeps the disc locked in world space so it never goes
        # edge-on and disappears mid-animation.
        rings_mesh = add_saturn_rings(planet)
        rings_mesh.rotation_mode = 'XYZ'
        add_rotation_driver(rings_mesh, -spn(spin_N), axis=2)

    if pname == "Uranus":
        rings_mesh = add_uranus_rings(planet)
        rings_mesh.rotation_mode = 'XYZ'
        add_rotation_driver(rings_mesh, -spn(spin_N), axis=2)

    add_location_drivers(planet, pdist, orb(orb_N), offset_angle=offset)
    add_rotation_driver(planet,         spn(spin_N), axis=2)

    planet_objs[pname] = planet

# ════════════════════════════════════════════════════════════════════════════
# ROCKETSHIP — launches from Earth, flies outward through the system.
# Parented to Earth so it inherits Earth's orbital motion as its launch point,
# then animates its LOCAL offset away from Earth (keyframed, not driver-based,
# since this is a one-time scripted event rather than a repeating cycle).
# ════════════════════════════════════════════════════════════════════════════

def make_emit_mat(name, color, strength=1.6):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out  = nt.nodes.new("ShaderNodeOutputMaterial")
    emit = nt.nodes.new("ShaderNodeEmission")
    emit.inputs[0].default_value = (*color, 1.0)
    emit.inputs[1].default_value = strength
    nt.links.new(emit.outputs[0], out.inputs[0])
    return mat


def build_rocketship(parent_planet, launch_frame_start, launch_frame_end,
                      local_launch_dir=(0, 0, 1), travel_distance=40.0):
    """
    Builds a simple composite rocketship (body, nose cone, fins) and parents
    it to parent_planet (Earth). It launches from the planet's surface and
    flies outward along local_launch_dir (in the planet's local space),
    covering travel_distance units over [launch_frame_start, launch_frame_end].
    Because it's parented, it automatically launches from wherever Earth is
    in its orbit at that point in the animation.
    """
    body_mat = make_emit_mat("Rocket_BodyMat", (0.88, 0.88, 0.90), 1.7)
    fin_mat  = make_emit_mat("Rocket_FinMat",  (0.75, 0.15, 0.12), 1.5)
    nose_mat = make_emit_mat("Rocket_NoseMat", (0.65, 0.68, 0.72), 1.7)

    length = 0.55
    radius = 0.09

    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=length, vertices=24,
                                         location=(0, 0, 0))
    body = bpy.context.object
    body.name = "Rocket_Body"
    body.data.materials.append(body_mat)
    bpy.ops.object.shade_smooth()

    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0.0, depth=length * 0.45,
                                     location=(0, 0, length / 2 + length * 0.22))
    nose = bpy.context.object
    nose.name = "Rocket_Nose"
    nose.data.materials.append(nose_mat)
    bpy.ops.object.shade_smooth()
    nose.parent = body

    for i, ang in enumerate((0, 120, 240)):
        bpy.ops.mesh.primitive_cone_add(radius1=radius * 1.6, radius2=0.0,
                                         depth=length * 0.35,
                                         location=(0, 0, -length * 0.35))
        fin = bpy.context.object
        fin.name = f"Rocket_Fin_{i}"
        fin.scale = (1.0, 0.12, 1.0)
        fin.rotation_euler[2] = math.radians(ang)
        fin.location.x = math.cos(math.radians(ang)) * radius * 1.1
        fin.location.y = math.sin(math.radians(ang)) * radius * 1.1
        fin.data.materials.append(fin_mat)
        bpy.ops.object.shade_smooth()
        fin.parent = body

    bpy.ops.mesh.primitive_cone_add(radius1=radius * 0.7, radius2=0.0,
                                     depth=length * 0.6,
                                     location=(0, 0, -length * 0.5 - length * 0.25))
    flame = bpy.context.object
    flame.name = "Rocket_Flame"
    flame.rotation_euler[0] = math.radians(180)
    flame.data.materials.append(make_emit_mat("Rocket_FlameMat", (1.0, 0.55, 0.1), 5.0))
    bpy.ops.object.shade_smooth()
    flame.parent = body

    # Parent the whole rocket to Earth — it now lives in Earth's local space.
    body.parent = parent_planet
    body.matrix_parent_inverse = parent_planet.matrix_world.inverted()

    # Orient the rocket to point along its launch direction.
    import mathutils
    dir_vec = mathutils.Vector(local_launch_dir).normalized()
    rot = dir_vec.to_track_quat('Z', 'Y').to_matrix().to_4x4()
    body.matrix_local = rot

    # Hide before launch, appear at the surface, fly outward, then fade.
    surface_offset = mathutils.Vector(local_launch_dir).normalized() * (
        PLANETS_BY_NAME["Earth"][1] * 1.05)   # just above Earth's radius

    body.scale = (0, 0, 0)
    body.keyframe_insert(data_path="scale", frame=max(1, launch_frame_start - 1))

    body.location = surface_offset
    body.scale = (1, 1, 1)
    body.keyframe_insert(data_path="location", frame=launch_frame_start)
    body.keyframe_insert(data_path="scale",    frame=launch_frame_start)

    end_offset = dir_vec * travel_distance
    body.location = end_offset
    body.keyframe_insert(data_path="location", frame=launch_frame_end)

    # Fade out (scale down) shortly after reaching the end point.
    fade_out_frame = min(TOTAL_FRAMES, launch_frame_end + 20)
    body.keyframe_insert(data_path="scale", frame=launch_frame_end)
    body.scale = (0, 0, 0)
    body.keyframe_insert(data_path="scale", frame=fade_out_frame)

    return body

# COMETS — small icy nucleus + long, thin, fading tail. These sweep through
# on one-time straight diagonal paths, fading in and out, spaced one after
# another so only one is ever visible at a time.

def make_comet_nucleus_mat(name, color=(0.75, 0.85, 0.95)):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out   = nt.nodes.new("ShaderNodeOutputMaterial")
    emit  = nt.nodes.new("ShaderNodeEmission")
    noise = nt.nodes.new("ShaderNodeTexNoise")
    ramp  = nt.nodes.new("ShaderNodeValToRGB")
    tc    = nt.nodes.new("ShaderNodeTexCoord")
    noise.inputs["Scale"].default_value = 6.0
    cr = ramp.color_ramp
    cr.elements[0].color = (color[0]*0.5, color[1]*0.5, color[2]*0.6, 1)
    cr.elements[1].color = (color[0]*1.3, color[1]*1.3, color[2]*1.3, 1)
    nt.links.new(tc.outputs["Generated"], noise.inputs["Vector"])
    nt.links.new(noise.outputs["Fac"], ramp.inputs["Fac"])
    nt.links.new(ramp.outputs["Color"], emit.inputs["Color"])
    emit.inputs["Strength"].default_value = 2.2
    nt.links.new(emit.outputs["Emission"], out.inputs["Surface"])
    return mat


def make_comet_tail_mat(name, color=(0.65, 0.85, 1.0)):
    """Thin, wispy, fast-fading tail — bright only near the nucleus."""
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    mat.blend_method = 'BLEND'
    nt = mat.node_tree
    for n in list(nt.nodes): nt.nodes.remove(n)
    out   = nt.nodes.new("ShaderNodeOutputMaterial")
    emit  = nt.nodes.new("ShaderNodeEmission")
    trans = nt.nodes.new("ShaderNodeBsdfTransparent")
    mix   = nt.nodes.new("ShaderNodeMixShader")
    tc    = nt.nodes.new("ShaderNodeTexCoord")
    ramp  = nt.nodes.new("ShaderNodeValToRGB")

    cr = ramp.color_ramp
    cr.elements[0].position = 0.0
    cr.elements[0].color = (0, 0, 0, 1)
    cr.elements[1].position = 0.55
    cr.elements[1].color = (1, 1, 1, 1)

    emit.inputs["Color"].default_value = (*color, 1)
    emit.inputs["Strength"].default_value = 3.0

    nt.links.new(tc.outputs["Generated"], ramp.inputs["Fac"])
    nt.links.new(ramp.outputs["Color"], mix.inputs["Fac"])
    nt.links.new(trans.outputs["BSDF"], mix.inputs[1])
    nt.links.new(emit.outputs["Emission"], mix.inputs[2])
    nt.links.new(mix.outputs["Shader"], out.inputs["Surface"])
    return mat


def spawn_comet(name, start, end, nucleus_radius, frame_start, frame_end, fade_frames=24):
    """Small nucleus + thin fading tail, flying start->end, fading in/out via
    emission strength, hidden (scaled to zero) outside its active window."""
    bpy.ops.mesh.primitive_ico_sphere_add(radius=nucleus_radius, subdivisions=3,
                                           location=(0, 0, 0))
    nucleus = bpy.context.object
    nucleus.name = name
    nucleus.visible_shadow = False

    mod = nucleus.modifiers.new("Displace", type='DISPLACE')
    tex = bpy.data.textures.new(name + "_disp", type='DISTORTED_NOISE')
    tex.noise_scale = 0.5
    mod.texture = tex
    mod.strength = 0.3 * nucleus_radius
    bpy.ops.object.shade_smooth()
    nucleus_mat = make_comet_nucleus_mat(name + "_NucleusMat")
    nucleus.data.materials.append(nucleus_mat)

    tail_len = nucleus_radius * 22
    bpy.ops.mesh.primitive_cone_add(radius1=nucleus_radius * 0.45, radius2=0.0,
                                     depth=tail_len)
    tail = bpy.context.object
    tail.name = name + "_Tail"
    tail.visible_shadow = False
    tail_mat = make_comet_tail_mat(name + "_TailMat")
    tail.data.materials.append(tail_mat)
    bpy.ops.object.shade_smooth()

    tail.parent = nucleus
    tail.location = (0, 0, -tail_len * 0.5)
    tail.rotation_euler[0] = math.radians(180)

    nucleus_emit = nucleus_mat.node_tree.nodes["Emission"]
    tail_emit    = tail_mat.node_tree.nodes["Emission"]
    nucleus_full = nucleus_emit.inputs["Strength"].default_value
    tail_full    = tail_emit.inputs["Strength"].default_value

    fade_in_end    = min(frame_end, frame_start + fade_frames)
    fade_out_start = max(frame_start, frame_end - fade_frames)

    for node, full in ((nucleus_emit, nucleus_full), (tail_emit, tail_full)):
        node.inputs["Strength"].default_value = 0.0
        node.inputs["Strength"].keyframe_insert(data_path="default_value", frame=frame_start)
        node.inputs["Strength"].default_value = full
        node.inputs["Strength"].keyframe_insert(data_path="default_value", frame=fade_in_end)
        node.inputs["Strength"].keyframe_insert(data_path="default_value", frame=fade_out_start)
        node.inputs["Strength"].default_value = 0.0
        node.inputs["Strength"].keyframe_insert(data_path="default_value", frame=frame_end)

    nucleus.scale = (0, 0, 0)
    nucleus.keyframe_insert(data_path="scale", frame=max(1, frame_start - 1))
    nucleus.scale = (1, 1, 1)
    nucleus.keyframe_insert(data_path="scale", frame=frame_start)
    nucleus.keyframe_insert(data_path="scale", frame=frame_end)
    nucleus.scale = (0, 0, 0)
    nucleus.keyframe_insert(data_path="scale", frame=min(TOTAL_FRAMES, frame_end + 1))

    import mathutils
    start_vec = mathutils.Vector(start)
    end_vec   = mathutils.Vector(end)
    direction = end_vec - start_vec
    rot_mat   = direction.to_track_quat('Z', 'Y').to_matrix().to_4x4()
    nucleus.matrix_world = mathutils.Matrix.Translation(start_vec) @ rot_mat
    nucleus.keyframe_insert(data_path="rotation_euler", frame=frame_start)

    span = max(frame_end - frame_start, 1)
    for f in (frame_start, frame_end):
        t = (f - frame_start) / span
        nucleus.location.x = start[0] + (end[0] - start[0]) * t
        nucleus.location.y = start[1] + (end[1] - start[1]) * t
        nucleus.location.z = start[2] + (end[2] - start[2]) * t
        nucleus.keyframe_insert(data_path="location", frame=f)

    return nucleus
